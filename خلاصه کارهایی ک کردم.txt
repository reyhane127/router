#run_ping()
def run_ping(self, target):

    command = [
        "ping",
        "-c", "4",
        "-W", str(config.PING_TIMEOUT),
        target
    ]

    result = subprocess.run(
        command,
        capture_output=True,
        text=True
    )

    return result

#check_ping()
#فقط بررسی می‌کند آیا لینک پاسخ می‌دهد یا نه.
def check_ping(self, ping_result):

    if ping_result.returncode == 0:
        return True

    return False


#measure_latency()
# زمان پاسخ (Ping) را اندازه می‌گیرد.

def measure_latency(self, ping_result):

    if ping_result.returncode != 0: #پینگ شکست خورده پس دیگ اصلا مهم نیس
        return None

    output = ping_result.stdout #کل پینگ ه اینجاست stdout

    match = re.search(    #به کمک ریجکس ها زمان پینگ هارو میگیریم ک میانگینشون رو حساب کنیم و بدیم بیرون
        r"=\s*([\d.]+)/([\d.]+)/([\d.]+)/([\d.]+)",
        output
    )

    if match:
        return float(match.group(2))

    return None




#measure_packet_loss()
    def measure_packet_loss(self, ping_result):

    if ping_result.returncode != 0:
        return 100.0

    output = ping_result.stdout

    match = re.search(
        r"(\d+)% packet loss",
        output
    )

    if match:
        return float(match.group(1))

    return None

#check_link()
#همه رو جمع میکنه یجا گزارش میده کار دیگ ای نمیکنه
def check_link(self, target):

    ping_result = self.run_ping(target)

    alive = self.check_ping(ping_result)

    latency = self.measure_latency(ping_result)

    packet_loss = self.measure_packet_loss(ping_result)

    dns=self.check_dns(config.DNS_TARGETS)

    http=self.check_http(config.HTTP_TARGETS)
 
    return {
        "alive": alive,
        "latency": latency,
        "packet_loss": packet_loss
        "dns" : dns
        "http" : http
    }


#check_dns()
#چک میکنه ک ایا dns  هایی ک از کانفیگ اضافه کردیم سالمن یا نه
def check_dns(self,targets):
    for target in config.DNS_TARGETS:

        result = subprocess.run(
            ["nslookup", target],
            capture_output=True,
            text=True
        )

        if result.returncode == 0:
            return True

    return False

#check_http
def check_http(self, targets):

    for target in targets:

        result = subprocess.run(
            [
                "curl",
                "-I",
                "--max-time",
                "3",
                target
            ],
            capture_output=True,
            text=True
        )

        if result.returncode == 0:
            return True

    return False
