#run_ping()
def run_ping(self, target):

    command = [
        "ping",
        "-c", "4",
        "-W", str(config.PING_TIMEOUT),
        target
    ]

    result = subprocess.run(
        command,
        capture_output=True,
        text=True
    )

    return result


#check_ping()
#فقط بررسی می‌کند آیا لینک پاسخ می‌دهد یا نه.
def check_ping(self, ping_result):

    if ping_result.returncode == 0:
        return True

    return False


#measure_latency()
# زمان پاسخ (Ping) را اندازه می‌گیرد.

def measure_latency(self, ping_result):

    if ping_result.returncode != 0: #پینگ شکست خورده پس دیگ اصلا مهم نیس
        return None

    output = ping_result.stdout #کل پینگ ه اینجاست stdout

    match = re.search(    #به کمک ریجکس ها زمان پینگ هارو میگیریم ک میانگینشون رو حساب کنیم و بدیم بیرون
        r"=\s*([\d.]+)/([\d.]+)/([\d.]+)/([\d.]+)",
        output
    )

    if match:
        return float(match.group(2))

    return None




    #measure_packet_loss()
    def measure_packet_loss(self, ping_result):

    if ping_result.returncode != 0:
        return 100.0

    output = ping_result.stdout

    match = re.search(
        r"(\d+)% packet loss",
        output
    )

    if match:
        return float(match.group(1))

    return None
